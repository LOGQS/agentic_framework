Resilience

Retry logic and rate limiting for robust agent execution

Overview

Agentic provides universal resilience utilities that work with any async operation. These utilities help build production-ready agents that gracefully handle transient failures and respect rate limits.

Retry Logic

The with_retry function wraps any async callable with automatic retry behavior:

from agentic import with_retry, RetryConfig

async def call_llm_api(prompt):
    # Potentially failing API call
    response = await llm_client.generate(prompt)
    return response

# Wrap with retry logic
resilient_llm_call = with_retry(
    call_llm_api,
    config=RetryConfig(
        max_attempts=3,
        initial_delay=1.0,
        max_delay=10.0,
        exponential_base=2.0,
        jitter=True
    )
)

# Use it like the original function
response = await resilient_llm_call("Generate a response")

RetryConfig Options

max_attempts

Maximum number of attempts (including the initial call):

RetryConfig(max_attempts=5)  # Try up to 5 times

initial_delay

Initial delay in seconds before first retry:

RetryConfig(initial_delay=2.0)  # Wait 2 seconds before retry

max_delay

Maximum delay between retries (prevents exponential growth from becoming too large):

RetryConfig(max_delay=60.0)  # Cap at 60 seconds

exponential_base

Base for exponential backoff calculation:

RetryConfig(exponential_base=2.0)  # Delay doubles each attempt

Delay formula: min(initial_delay * (exponential_base ** attempt), max_delay)

jitter

Add randomness to delays to prevent thundering herd:

RetryConfig(jitter=True)  # Add random variation to delays

retryable_exceptions

Tuple of exception types that should trigger a retry:

from aiohttp import ClientError, ServerTimeoutError

RetryConfig(
    retryable_exceptions=(ClientError, ServerTimeoutError, TimeoutError)
)

on_retry

Callback function called before each retry:

def log_retry(attempt, exception, delay):
    print(f"Retry {attempt}: {exception}. Waiting {delay}s...")

RetryConfig(on_retry=log_retry)

Common Retry Patterns

API Calls with Retry

from agentic import with_retry, RetryConfig
import aiohttp

async def fetch_data(url):
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as response:
            return await response.json()

# Wrap with retry
resilient_fetch = with_retry(
    fetch_data,
    RetryConfig(
        max_attempts=5,
        initial_delay=1.0,
        exponential_base=2.0,
        retryable_exceptions=(
            aiohttp.ClientError,
            asyncio.TimeoutError
        )
    )
)

data = await resilient_fetch("https://api.example.com/data")

Database Operations

async def save_to_db(record):
    await db.insert(record)

resilient_save = with_retry(
    save_to_db,
    RetryConfig(
        max_attempts=3,
        initial_delay=0.5,
        retryable_exceptions=(ConnectionError, TimeoutError)
    )
)

await resilient_save({"id": 1, "data": "..."})

LLM Provider Integration

class ResilientLLMProvider:
    def __init__(self, base_provider):
        self._provider = base_provider

        # Wrap generate with retry
        self._resilient_generate = with_retry(
            self._provider.generate,
            RetryConfig(
                max_attempts=3,
                initial_delay=2.0,
                exponential_base=2.0,
                jitter=True
            )
        )

    def generate(self, prompt, **kwargs):
        return self._resilient_generate(prompt, **kwargs)

    async def stream(self, prompt, **kwargs):
        # Stream method with retry
        resilient_stream = with_retry(
            self._provider.stream,
            RetryConfig(max_attempts=2)
        )
        async for chunk in await resilient_stream(prompt, **kwargs):
            yield chunk

Rate Limiting

The RateLimiter class implements token bucket rate limiting:

from agentic import RateLimiter

# Create rate limiter: 10 requests per second
limiter = RateLimiter(
    rate=10.0,        # Tokens per second
    capacity=10.0     # Bucket capacity (burst allowance)
)

# Acquire token before operation
await limiter.acquire()  # Blocks until token available
await call_api()

Rate Limiter Configuration

Basic Rate Limiting

# 5 requests per second
limiter = RateLimiter(rate=5.0, capacity=5.0)

for i in range(10):
    await limiter.acquire()
    await make_request(i)

Burst Allowance

Higher capacity allows bursts while maintaining average rate:

# 10 requests/sec average, allow bursts up to 50
limiter = RateLimiter(rate=10.0, capacity=50.0)

# Can handle 50 immediate requests
# Then throttles to 10/sec

Conservative Rate Limiting

Lower capacity enforces stricter pacing:

# 10 requests/sec, no bursting
limiter = RateLimiter(rate=10.0, capacity=1.0)

# Forces even distribution over time

Combining Retry and Rate Limiting

from agentic import with_retry, RetryConfig, RateLimiter

# Rate limiter for API calls
api_limiter = RateLimiter(rate=10.0, capacity=10.0)

async def call_api_with_limits(endpoint, data):
    # Wait for rate limit token
    await api_limiter.acquire()

    # Make API call
    response = await api_client.post(endpoint, json=data)
    return response.json()

# Add retry logic on top of rate limiting
resilient_api_call = with_retry(
    call_api_with_limits,
    RetryConfig(
        max_attempts=3,
        initial_delay=1.0,
        exponential_base=2.0
    )
)

# Use the resilient, rate-limited function
result = await resilient_api_call("/endpoint", {"key": "value"})

Advanced Patterns

Per-Agent Rate Limiting

class RateLimitedAgent:
    def __init__(self, agent, rate_limit):
        self._agent = agent
        self._limiter = RateLimiter(
            rate=rate_limit,
            capacity=rate_limit
        )

    async def step(self, user_input):
        # Rate limit agent execution
        await self._limiter.acquire()

        runner = AgentRunner(self._agent)
        async for event in runner.step_stream(user_input):
            yield event

# Create rate-limited agent (max 2 steps per second)
limited_agent = RateLimitedAgent(my_agent, rate_limit=2.0)

Conditional Retry

def should_retry_on_status(exception):
    # Only retry on specific HTTP status codes
    if isinstance(exception, aiohttp.ClientResponseError):
        return exception.status in (429, 500, 502, 503, 504)
    return True

resilient_call = with_retry(
    api_function,
    RetryConfig(
        max_attempts=5,
        retryable_exceptions=(aiohttp.ClientError,),
        on_retry=lambda attempt, exc, delay:
            should_retry_on_status(exc)
    )
)

Circuit Breaker Pattern

class CircuitBreaker:
    def __init__(self, failure_threshold=5, recovery_timeout=60.0):
        self._failures = 0
        self._threshold = failure_threshold
        self._timeout = recovery_timeout
        self._last_failure = None
        self._is_open = False

    async def call(self, func, *args, **kwargs):
        # Check if circuit is open
        if self._is_open:
            if time.time() - self._last_failure < self._timeout:
                raise RuntimeError("Circuit breaker is open")
            else:
                # Try to close circuit
                self._is_open = False
                self._failures = 0

        try:
            result = await func(*args, **kwargs)
            self._failures = 0  # Reset on success
            return result
        except Exception as e:
            self._failures += 1
            self._last_failure = time.time()

            if self._failures >= self._threshold:
                self._is_open = True
                print("Circuit breaker opened!")

            raise e

breaker = CircuitBreaker(failure_threshold=5, recovery_timeout=30.0)
result = await breaker.call(unreliable_api_call)

Monitoring Retries

Retry Metrics

class RetryMetrics:
    def __init__(self):
        self.total_attempts = 0
        self.successful_retries = 0
        self.failed_operations = 0

    def on_retry(self, attempt, exception, delay):
        self.total_attempts += 1
        print(f"Retry {attempt}: {exception.__class__.__name__}")

    def on_success(self, attempts):
        if attempts > 1:
            self.successful_retries += 1

    def on_failure(self):
        self.failed_operations += 1

metrics = RetryMetrics()

resilient_op = with_retry(
    operation,
    RetryConfig(
        max_attempts=3,
        on_retry=metrics.on_retry
    )
)

try:
    result = await resilient_op()
    metrics.on_success(attempts_used)
except Exception:
    metrics.on_failure()

print(f"Total retry attempts: {metrics.total_attempts}")
print(f"Operations recovered via retry: {metrics.successful_retries}")
print(f"Failed operations: {metrics.failed_operations}")

Best Practices

  • Set Appropriate Timeouts: Always set max_delay to prevent excessively long waits
  • Use Jitter: Enable jitter in production to prevent thundering herd
  • Limit Retry Attempts: Don't retry indefinitely - set reasonable max_attempts
  • Log Retries: Use on_retry callback to monitor retry behavior
  • Retry Transient Errors Only: Only retry errors that might succeed on retry
  • Respect Rate Limits: Use RateLimiter to stay within API rate limits
  • Combine Strategies: Use retry with rate limiting for robust operations
  • Monitor Metrics: Track retry counts to identify unstable services
  • Circuit Breakers: Implement circuit breakers for frequently failing operations
Warning: Excessive retries can amplify failures during outages. Always set max_attempts and use exponential backoff with jitter.

Real-World Example

from agentic import with_retry, RetryConfig, RateLimiter
import logging

logger = logging.getLogger("agent.resilience")

# Global rate limiter for API
api_limiter = RateLimiter(rate=50.0, capacity=100.0)

def log_retry(attempt, exception, delay):
    logger.warning(
        f"LLM API retry {attempt}",
        extra={
            "error": str(exception),
            "delay": delay
        }
    )

class ProductionLLMProvider:
    def __init__(self, base_provider):
        self._provider = base_provider

        # Wrap generate with full resilience
        self._resilient_generate = with_retry(
            self._rate_limited_generate,
            RetryConfig(
                max_attempts=5,
                initial_delay=1.0,
                max_delay=30.0,
                exponential_base=2.0,
                jitter=True,
                retryable_exceptions=(
                    ConnectionError,
                    TimeoutError,
                    asyncio.TimeoutError
                ),
                on_retry=log_retry
            )
        )

    async def _rate_limited_generate(self, prompt, **kwargs):
        # Acquire rate limit token
        await api_limiter.acquire()

        # Call underlying provider
        return await self._provider.generate(prompt, **kwargs)

    async def generate(self, prompt, **kwargs):
        return await self._resilient_generate(prompt, **kwargs)

# Use in production
production_provider = ProductionLLMProvider(base_llm_provider)
agent = Agent(config, context, patterns, tools, production_provider)
